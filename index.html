Portainer Quick-Stack

Deploy directly in your home lab using Portainer or Docker Compose:

version: '3.8'
services:
  fhem-mcp:
    image: ghcr.io/jbkunama1/hai.mcpservers/fhem-mcp:latest
    container_name: fhem-mcp
    restart: unless-stopped
    environment:
      - MCP_API_KEY=your-secret-key
      - FHEM_URL=http://192.168.178.15:8085/fhem
    ports:
      - "5887:8000"
    networks:
      - highfishNetwork
